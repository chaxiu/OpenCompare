 *
 * Two trees are scanned through a directory capability, paired by relative
 * path and classified into the product's four file semantics:
 * `same` / `added` (right only) / `removed` (left only) / `modified`.
 *
 * The fast path never reads file bodies: a size mismatch is immediately
 * `modified`, an mtime mismatch (at millisecond granularity) is immediately
 * `modified`, and only pairs whose size *and* mtime agree fall through to a
 * SHA-256 content confirmation (which is what catches "same size, different
 * content"). Hashes are cached per `(path, size, modifiedMs)`, so re-comparing
 * an unchanged tree neither reads bodies nor recomputes digests.

  readonly kind: 'file' | 'directory'
  readonly size: number
  readonly modifiedMs: number

  readonly linkTarget?: string

/**

  readonly phaseFiles: number
  /** Bytes settled inside the current phase. */
  readonly phaseBytes: number
}

/**

 *
 * A real pair (DawnChat ↔ ZenMind) settles around 78k equal-size/mtime pairs,
 * so the entry cap leaves room for one full comparison plus its warm rounds
 * while still bounding a process that opens many unrelated trees. The byte cap
 * is the second, independent bound: an entry is `path + digest` text plus a
 * fixed bookkeeping overhead, so a deep tree with long paths hits the byte
 * budget before the entry count does.
 */

export const DEFAULT_COMPARISON_CACHE_MAX_BYTES = 128 * 1024 * 1024

 * documented in `docs/perf/large-directory-memory.md` — not a heap measurement.
 */

/**

 * tests read it directly; its iteration order is the recency order (oldest
 * first), and every writer goes through {@link ComparisonCache.set} so the
 * byte accounting cannot drift. A write that does not fit the budget evicts
 * the least recently used entries — never the whole cache, and never the entry

  /** LRU write: refreshes recency, charges bytes and evicts down to the caps. */

  delete(path: string): void

   */

   * session is closed. Returns how many entries were released.

   * what keeps the cross-comparison memo bounded.

function mergeChildren(
  relativePath: string,
  left: ScannedNode,
  right: ScannedNode,
  context: CompareContext

function buildNode(
  relativePath: string,
  left: ScannedNode | undefined,
  right: ScannedNode | undefined,
  context: CompareContext
): DirectoryDiffNode {

  const bothFiles = left?.kind === 'file' && right?.kind === 'file'
 
  if (bothFiles) {
    const status = classifyPair(left, right, context)

      relativePath,
      kind: 'file',
      status,
      children: [],
      counts: countEntry(status),
      leftPath: left.path,
      rightPath: right.path,
      leftSize: left.size,
      rightSize: right.size,
      leftModifiedMs: left.modifiedMs,
      rightModifiedMs: right.modifiedMs,
      leftIsSymbolicLink: left.linkTarget !== undefined,
      rightIsSymbolicLink: right.linkTarget !== undefined,
      leftLinkTarget: left.linkTarget,
      rightLinkTarget: right.linkTarget,
      linkTargetKind: left.linkTargetKind ?? right.linkTargetKind
    }
  }
 
  // A symbolic link against a directory on the other side is one type-clash
  // entry: the link is a leaf, so the directory subtree is never flattened in.
  if (left && right && (left.linkTarget !== undefined || right.linkTarget !== undefined)) {
    return {
      name,
      relativePath,
      kind: 'file',
      status: 'modified',
      children: [],
      counts: countEntry('modified'),
      leftPath: left.path,
      rightPath: right.path,
      leftSize: left.kind === 'file' ? left.size : undefined,
      rightSize: right.kind === 'file' ? right.size : undefined,
      leftModifiedMs: left.kind === 'file' ? left.modifiedMs : undefined,
      rightModifiedMs: right.kind === 'file' ? right.modifiedMs : undefined,
      leftIsSymbolicLink: left.linkTarget !== undefined,
      rightIsSymbolicLink: right.linkTarget !== undefined,
      leftLinkTarget: left.linkTarget,
      rightLinkTarget: right.linkTarget,
      linkTargetKind: left.linkTargetKind ?? right.linkTargetKind
    }
  }
 
  if (left && right) {
    const children = mergeChildren(relativePath, left, right, context)
    const counts = aggregate(children)
    // A file/directory type clash is reported as a single modified directory.
    const sameKind = left.kind === 'directory' && right.kind === 'directory'
    return {
      name,
      relativePath,
      kind: 'directory',
      status: sameKind ? directoryStatus(counts) : 'modified',
      children,
      counts,
      leftPath: left.path,
      rightPath: right.path
    }
  }
 
  const only = (left ?? right) as ScannedNode
  return markSubtree(relativePath, only, left ? 'removed' : 'added', left ? 'left' : 'right')
}

/**

 * mtime fast path is inconclusive. Bodies are read through a bounded window so a
 * multi-thousand-file tree cannot exhaust the process file descriptors.
 */
export async function compareDirectories(
  leftRoot: string,
  rightRoot: string,
  list: ListDirectory,
  readTextFile: ReadTextFile,
  cache: ComparisonCache,
  /** Digest implementation; defaults to the built-in synchronous SHA-256. */
  hash?: (text: string) => string,
  /** In-flight body-read window and optional digest capability. */
  options: ComparisonIoOptions = {}

  // every pair; the caps apply again the moment this comparison ends.
  const releaseHold = cache.hold()

    // Both side scans feed one monotonic progress stream.
    const sideProgress = createScanProgressMerger(options.onProgress)
    const [left, right] = await Promise.all([
      scanDirectory(leftRoot, list, { ...options, onProgress: sideProgress.reporter() }),
      scanDirectory(rightRoot, list, { ...options, onProgress: sideProgress.reporter() })
    ])
 
    // Digests go straight into the cache: only one body per in-flight read is
    // held in memory, which is what lets two real project trees be compared.
    // Their progress continues from the closing scan totals.
    const unreadable = await digestContentCandidates(
      left,

}
